% Name of the chord-chart coordinate at the center of bar #1, for add.tikz:

% --- Page position of every bar, for the apk -------------------------------
% A song whose main.tex defines \writesongcells (books do not) writes cells.yml
% next to its PDF: for each BAR_<n> of song.tikz, the cell it sits on, in PDF
% points (1/72 inch, as in the PDF) from the top-left corner of the page. song.tikz calls \recordbar as
% it defines the coordinates, and \songplacechart, which scales the chart box
% to the text width, converts them to page positions.
\newsavebox{\songchartbox}
\newwrite\songcellsfile
\newdimen\songdx \newdimen\songdy \newdimen\songdw \newdimen\songdh
\gdef\songcellrecords{}
\gdef\cursection{}
\gdef\songminx{0pt}
\gdef\songminy{0pt}
\makeatletter
% \recordbar{n}{w}{h}{chords}{lyric}: remember the center of BAR_n and the size of its cell
% (w and h are pgfmath expressions, in cm), its chords (ASCII, space separated) and the
% syllable sung on its first beat (detokenized text).
\newcommand*{\recordbar}[5]{%
  \ifdefined\writesongcells
    \pgfpointanchor{BAR_#1}{center}%
    \pgf@xa=\pgf@x \pgf@ya=\pgf@y
    \pgfmathsetlength\pgf@xb{(#2)*1cm}%
    \pgfmathsetlength\pgf@yb{(#3)*1cm}%
    \xdef\songcellrecords{\unexpanded\expandafter{\songcellrecords}\noexpand\songcellrecord%
      {#1}{\the\pgf@xa}{\the\pgf@ya}{\the\pgf@xb}{\the\pgf@yb}%
      {\detokenize\expandafter{\cursection}}{#4}{#5}}%
  \fi}
% \recorddot{first}{last}: the dot of a repeat marker drawn at the coordinate REPDOT, for the
% passes of the repeated block whose bars are first..last.
\gdef\songdotrecords{}
\newcommand*{\recorddot}[2]{%
  \ifdefined\writesongcells
    \pgfpointanchor{REPDOT}{center}%
    \pgf@xa=\pgf@x \pgf@ya=\pgf@y
    \xdef\songdotrecords{\unexpanded\expandafter{\songdotrecords}\noexpand\songdotrecord%
      {#1}{#2}{\the\pgf@xa}{\the\pgf@ya}}%
  \fi}
% Bounding box of the picture, read just before TikZ closes it.
\newcommand*{\songendpicture}{%
  \xdef\songminx{\the\pgf@picminx}\xdef\songminy{\the\pgf@picminy}}
% \songcellrecord{n}{x}{y}{w}{h}{section}{chords}{lyric}: x y is the center in the picture,
% w h the size, section the title of the section the bar belongs to, chords the chords of the bar.
% Written with the position \songplacechart saved with \savepos.
\newcommand*{\songcellrecord}[8]{%
  \pgfmathsetlength\songdx{\songscale*(#2-\songminx-#4/2)}%
  \pgfmathsetlength\songdy{\songscale*(#3-\songminy+#5/2)}%
  \pgfmathsetlength\songdw{\songscale*#4}%
  \pgfmathsetlength\songdh{\songscale*#5}%
  \edef\songwrite{\noexpand\write\songcellsfile{%
    - bar: #1^^J%
    \space\space page: \noexpand\thepage^^J%
    \space\space x: \noexpand\strip@pt\noexpand\dimexpr(\noexpand\lastxpos sp+\the\songdx)*7200/7227\noexpand\relax^^J%
    \space\space y: \noexpand\strip@pt\noexpand\dimexpr(\the\paperheight-\noexpand\lastypos sp-\the\songdy)*7200/7227\noexpand\relax^^J%
    \space\space w: \strip@pt\dimexpr\songdw*7200/7227\relax^^J%
    \space\space h: \strip@pt\dimexpr\songdh*7200/7227\relax^^J%
    \space\space section: "#6"^^J%
    \space\space chords: "#7"^^J%
    \space\space lyric: "#8"}}%
  \songwrite}
% \songdotrecord{first}{last}{x}{y}: x y is the center of the dot in the picture. Written like the cells.
\newcommand*{\songdotrecord}[4]{%
  \pgfmathsetlength\songdx{\songscale*(#3-\songminx)}%
  \pgfmathsetlength\songdy{\songscale*(#4-\songminy)}%
  \pgfmathsetlength\songdw{\songscale*0.1cm}%
  \edef\songwrite{\noexpand\write\songcellsfile{%
    - first: #1^^J%
    \space\space last: #2^^J%
    \space\space page: \noexpand\thepage^^J%
    \space\space x: \noexpand\strip@pt\noexpand\dimexpr(\noexpand\lastxpos sp+\the\songdx)*7200/7227\noexpand\relax^^J%
    \space\space y: \noexpand\strip@pt\noexpand\dimexpr(\the\paperheight-\noexpand\lastypos sp-\the\songdy)*7200/7227\noexpand\relax^^J%
    \space\space r: \strip@pt\dimexpr\songdw*7200/7227\relax}}%
  \songwrite}
\newcommand*{\songplacechart}{%
  \ifdefined\writesongcells
    \pgfmathsetmacro{\songscale}{min(1,\the\textwidth/\the\wd\songchartbox)}%
    \immediate\openout\songcellsfile=cells.yml
    \immediate\write\songcellsfile{page_width_pt: \strip@pt\dimexpr\paperwidth*7200/7227\relax^^J%
      page_height_pt: \strip@pt\dimexpr\paperheight*7200/7227\relax^^Jcells:}%
    \noindent\savepos
    \scalebox{\songscale}{\usebox{\songchartbox}}%
    \songcellrecords
    \write\songcellsfile{dots:}%
    \songdotrecords
    \write\songcellsfile{syllables:}%
    \gdef\songcellrecords{}%
    \gdef\songdotrecords{}%
  \else
    \pgfmathsetmacro{\songscale}{min(1,\the\textwidth/\the\wd\songchartbox)}%
    \noindent\scalebox{\songscale}{\usebox{\songchartbox}}%
  \fi}
\makeatother

% In the lyrics of a song that writes cells.yml, the k-th \songwordfb of a section is its k-th
% bar: \songsylstart{first bar of the section} (before the lyrics of a section whose
% \songwordfb match its bars) and \songsylstop. The position of each non-empty one goes to
% cells.yml, under syllables:.
\newcounter{songsylbar}
\newif\ifsongsyl
\newsavebox{\songsylbox}
\newcommand*{\songsylstart}[1]{\setcounter{songsylbar}{#1}\addtocounter{songsylbar}{-1}\songsyltrue}
\newcommand*{\songsylstop}{\songsylfalse}
\makeatletter
% \songrecordsyl{text}: typeset text in the box and write its page position (top left, size, PDF points)
\newcommand*{\songrecordsyl}[1]{%
  \sbox{\songsylbox}{\fcolorbox{black}{red!20}{#1}}%
  \savepos
  \edef\songwrite{\noexpand\write\songcellsfile{%
    - bar: \thesongsylbar^^J%
    \space\space page: \noexpand\thepage^^J%
    \space\space x: \noexpand\strip@pt\noexpand\dimexpr\noexpand\lastxpos sp*7200/7227\noexpand\relax^^J%
    \space\space y: \noexpand\strip@pt\noexpand\dimexpr(\the\paperheight-\noexpand\lastypos sp-\the\ht\songsylbox)*7200/7227\noexpand\relax^^J%
    \space\space w: \strip@pt\dimexpr\wd\songsylbox*7200/7227\relax^^J%
    \space\space h: \strip@pt\dimexpr(\ht\songsylbox+\dp\songsylbox)*7200/7227\relax}}%
  \songwrite
  \usebox{\songsylbox}}
\makeatother
