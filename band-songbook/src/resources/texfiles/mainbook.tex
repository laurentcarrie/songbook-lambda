% Master file of a book: collates the songs listed in the book yaml.
% Each song is imported from its own build directory, so the \input commands
% inside body.tex (song.tikz, lyrics/*.tex) resolve in that directory.
\documentclass{article}
\PassOptionsToPackage{x11names}{xcolor}
\usepackage{tikz}
\usepackage{import}
\input{preamble}
\input{chords}
\input{sections}
% Room under the body for the navigation line, which sits above the foot rule:
% the default footskip leaves it grazing the last line of a page full of lyrics.
\geometry{footskip=46pt}
\begin{document}

% The setlist is an enumerate rather than a tabular or a titlepage: a book of
% any length keeps its numbering and breaks over as many pages as it needs.
\thispagestyle{empty}
\phantomsection\label{book:top}
\begin{center}
	{
		\songbookauthorfont
		\fontsize{40pt}{40pt}\selectfont
		\color{blue}
		{{{book.name}}}
	}
\end{center}
\vspace{2em}
% The entries link to the page their song starts on. linkcolor is black for
% the whole document, so it is turned up for the setlist only, and put back
% afterwards: hypersetup does not respect grouping.
\hypersetup{linkcolor=blue}
{
	\fontsize{14pt}{22pt}\selectfont
	\begin{enumerate}
{{{setlist}}}
	\end{enumerate}
}
\hypersetup{linkcolor=black}
\clearpage

{{{songs_body}}}

\end{document}
